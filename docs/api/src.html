241    def _log_context_usage(self, response: ModelResponse) -> None:
242        usage = response.usage
243        if not usage or "input_tokens" not in usage:
244            log_info(
245                "CONTEXT_USAGE_UNAVAILABLE: provider did not return token usage; "
246                f"configured_context_window_tokens={self.context_window_tokens}"
247            )
248            return
249
250        input_tokens = usage.get("input_tokens", 0)
251        output_tokens = usage.get("output_tokens", 0)
252        context_tokens = input_tokens + output_tokens
253        remaining_tokens = self.context_window_tokens - context_tokens
254        utilization = context_tokens / self.context_window_tokens
255        message = (
256            "CONTEXT_USAGE: "
257            f"input_tokens={input_tokens} output_tokens={output_tokens} "
258            f"compared_context_tokens={context_tokens} "
259            f"context_window_tokens={self.context_window_tokens} "
260            f"remaining_tokens={remaining_tokens} utilization={utilization:.1%}"
261        )
262        if remaining_tokens <= 0:
263            log_warning(message + " status=over_context_window")
264        else:
265            log_info(message + " status=within_context_window")

268        if tool_call.parse_error:
269            return ToolResult(

271                output=(

273                    "Re-issue the call with a valid JSON object matching the tool schema."
274                ),
275                is_error=True,
276            )
277        if self.registry.get(tool_call.name) is None:
278            available = ", ".join(t.name for t in self.registry.all()) or "none"
279            return ToolResult(
280                call_id=tool_call.id,
281                output=f"Tool error: unknown tool '{tool_call.name}'. Available tools: {available}.",
282                is_error=True,
283            )
284
285        # Safety rules run first so no allow-list or callback can bypass them.
286        violation = self.permission_gate.safety_violation(tool_call.name, tool_call.arguments)
287        if violation:

289            return ToolResult(call_id=tool_call.id, output=f"Permission denied: {violation}", is_error=True)
290
291        if self.permission_callback:
292            allowed, reason = await self.permission_callback(tool_call.name, tool_call.arguments)
293        else:
294            allowed, reason = self.permission_gate.check(tool_call.name, tool_call.arguments)
295        if not allowed:
296            return ToolResult(
297                call_id=tool_call.id,
298                output=f"Permission denied: {reason}",
299                is_error=True,
300            )
301
302        if self.tool_callback:
303            await self.tool_callback(tool_call.name, tool_call.arguments, "started", False)
304
305        try:
306            output = await self.registry.execute(tool_call.name, **tool_call.arguments)
307            if self.tool_callback:
308                await self.tool_callback(tool_call.name, tool_call.arguments, output, False)
309            return ToolResult(call_id=tool_call.id, output=output, is_error=False)
310        except Exception as e:
311            detail = str(e).strip() or repr(e)
312            error_output = f"Tool error: {type(e).__name__}: {detail}"
313            if self.tool_callback:
314                await self.tool_callback(tool_call.name, tool_call.arguments, error_output, True)
315            return ToolResult(
316                call_id=tool_call.id,
317                output=error_output,
318                is_error=True,
319            )

142    async def normalize_input(self, user_input: str) -> str:
143        """Normalize user input end-to-end.
144
145        1. Refurbish the raw request via the model provider.
146        2. Split the result into objective + tasks.
147        3. Detect gaps in each task and ask the user (with suggestions) via ask_user.
148        4. Take user input once all gaps are filled.
149        5. Preserve the original input, refurbish the requirement, build a task object,
150           and return it all as the normalized user input string.
151        """
152        assert self.provider is not None
153        self.end_stream()
154
155        # Step 1: refurbish / normalize via the provider.
156        response = await self.provider.complete(
157            system=self.config.normalizer_system_prompt,
158            messages=[
159                Message.user(
160                    build_user_input_refurbish_prompt(
161                        user_input, user_input_refurbish_prompt=self.config.user_input_refurbish_prompt
162                    )
163                )
164            ],
165            tools=[],
166        )
167        normalized_text = "".join(
168            block.text for block in response.blocks if isinstance(block, TextBlock)
169        )
170        self.end_stream()
171        if normalized_text.startswith("Error:"):
172            self._ui.normalization_failed(normalized_text)
173            return normalized_text
174
175        # Non-interactive: return the refurbished text as-is.
176        if not self.interactive:
177            self._ui.input_normalized(normalized_text)
178            return normalized_text
179
180        # Step 2: split into objective + tasks.
181        objective, raw_tasks = self._split_normalized(normalized_text)
182        if not raw_tasks:
183            self._ui.input_normalized(normalized_text)
184            return normalized_text
185
186        # Step 3: build canonical task titles (strip any " — answer" enrichment
187        # from prior calls so we match against confirmed decisions).
188        tasks: list[str] = []
189        for t in raw_tasks:
190            canonical = self._canonical_task_title(t)
191            if canonical not in tasks:
192                tasks.append(canonical)
193
194        # Step 4: skip tasks whose gap was already resolved (don't re-ask,
195        # don't re-detect — prevents recursive gaps and duplicate questions).
196        pending_tasks = [t for t in tasks if t not in self.resolved_gaps]
197        if not pending_tasks:
198            self._ui.input_normalized(normalized_text)
199            return normalized_text
200
201        # Step 5: detect actual gaps only against the ORIGINAL objective and
202        # canonical task titles — never against gap descriptions, answers, or
203        # generated tasks (prevents recursive gaps).
204        gaps: list[dict[str, Any]] = []
205        for task in pending_tasks:
206            gap, suggestions = await self._detect_gap(objective, task)
207            if gap:
208                gaps.append({"task": task, "gap": gap, "suggestions": suggestions})
209
210        # Step 6: if no real gaps, continue with normalized input unchanged.
211        if not gaps:
212            self._ui.input_normalized(normalized_text)
213            return normalized_text
214
215        # Step 7: clarify each gap one-by-one with suggestions.
216        clarified_tasks: list[dict[str, str]] = []
217        for task in tasks:
218            matching_gap = None
219            for g in gaps:
220                if g["task"] == task:
221                    matching_gap = g
222                    break
223            if matching_gap is None:
224                clarified_tasks.append({"title": task, "status": "pending"})
225                continue
226
227            gap = matching_gap["gap"]
228            suggestions = matching_gap["suggestions"]
229            self._ui.console.print(f"\n[ui.subheader]Gap found for task:[/ui.subheader] {task}")
230            self._ui.console.print(f"[status.warning]Missing/unclear:[/status.warning] {gap}")
231            if suggestions:
232                self._ui.console.print(f"Suggestions: {', '.join(suggestions)}")
233
234            suggestion_text = ", ".join(suggestions) if suggestions else "please clarify"
235            question = (
236                f"Task '{task}' has a gap: {gap}. "
237                f"Did you mean {suggestion_text}? Or provide more details."
238            )
239            answer = await self.ask_user(question, suggestions or None)
240            # Preserve the confirmed decision keyed by canonical task title so
241            # this question is never asked again. Do NOT enrich the task title
242            # with the answer (prevents recursive gap detection).
243            self.confirmed_decisions[task] = (answer or "").strip()
244            self.resolved_gaps.add(task)
245            clarified_tasks.append({"title": task, "status": "pending"})
246
247        # Step 8: check existing skills/agents — internal only, never added as
248        # executable tasks (prevents scope expansion and task contamination).
249        matched_capabilities = await self._match_existing_capabilities(objective, tasks)
250        self._register_loaded_capabilities(matched_capabilities)
251
252        # Step 9: reference the normalized input with answers preserved in
253        # confirmed_decisions, but task titles stay clean.
254        final_text = self._assemble_normalized(
255            user_input, objective, clarified_tasks, matched_capabilities
256        )
257        self._ui.input_normalized(final_text)
258        return final_text

261        """Split normalized text into (objective, [tasks])."""
262        objective = ""
263        tasks: list[str] = []
264        marker = "Tasks to achieve:"
265        if marker in text:

274    async def _detect_gap(self, objective: str, task: str) -> tuple[str | None, list[str]]:
275        """Use the provider to detect gaps in a task and suggest clarifications.
276
277        Returns (gap_description_or_None, [suggestions]).
278        Only returns a gap if one actually exists; otherwise returns (None, []).
279        Skips entirely if the task's canonical title is already resolved
280        (prevents recursive gaps and duplicate questions).
281        """
282        if not self.interactive or self.provider is None:
283            return None, []
284
285        canonical = self._canonical_task_title(task)
286        if canonical in self.resolved_gaps:
287            return None, []
288
289        detect_prompt = (
290        "You are a requirement gap detector. Identify only missing information "
291        "that is essential to fulfilling the user's actual objective.\n\n"
292
293        f"Overall objective: {objective}\n"
294        f"Task to review: {canonical}\n\n"
295
296        "Rules:\n"
297        "- Detect only gaps that materially affect the requested outcome.\n"
298        "- Do not flag routine implementation, execution, or verification details.\n"
299        "- Do not flag information that can be reasonably inferred from context.\n"
300        "- Do not flag implementation choices the agent can decide with reasonable defaults.\n"
301        "- Do not suggest additional features or expand the task scope.\n"
302        "- Do not treat optional improvements or preferences as required information.\n"
303        "- If the user asks to create/edit a file but did not specify its content or "
304        "purpose, flag that as a GAP (do not invent content).\n"
305        "- If the task can be completed correctly using reasonable defaults, "
306        "consider it COMPLETE.\n"
307        "- Ask for clarification only when missing information could lead to "
308        "a materially incorrect result.\n\n"
309
310        "If no critical gap exists, reply exactly:\n"
311        "COMPLETE\n\n"
312
313        "If a critical gap exists, reply exactly:\n"
314        "GAP: <what is essential and missing>\n"
315        "SUGGEST: <recommended option> | <alternative 1> | <alternative 2>\n\n"
316
317        "Keep the response concise. Return only COMPLETE or the GAP format."
318    )
319        try:
320            response = await self.provider.complete(
321                system="You are a precise gap-detection assistant. Reply only in the requested format.",
322                messages=[Message.user(detect_prompt)],
323                tools=[],
324            )
325            result = "".join(
326                block.text for block in response.blocks if isinstance(block, TextBlock)
327            ).strip()
328        except Exception:
329            return None, []
330
331        if result == "COMPLETE" or not result:
332            return None, []
333        if result.startswith("GAP:"):
334            lines = result.splitlines()
335            gap = lines[0][len("GAP:"):].strip()
336            suggestions: list[str] = []
337            for line in lines[1:]:
338                if line.startswith("SUGGEST:"):
339                    raw = line[len("SUGGEST:"):]
340                    suggestions = [s.strip() for s in raw.split("|") if s.strip()]
341                    break
342            return gap, suggestions
343        return None, []

361    ) -> list[dict[str, Any]]:
362        """Check existing skills and agents (from agents.md) against the tasks.
363

365        complete each task. Returns a list of matches with type, name, and the
366        task indices they apply to.

368        if not self.interactive or self.provider is None:
369            return []
370
371        agents_md = self.agent_loader.read_agents_md_content() if self.agent_loader else ""
372
373        # Gather discovered skills and agents.
374        skills: list[Any] = []
375        agents: list[Any] = []
376        if self.skill_loader is not None:
377            try:

379            except Exception:
380                skills = []
381        if self.agent_loader is not None:
382            try:
383                agents = self.agent_loader.all()
384            except Exception:
385                agents = []
386
387        if not skills and not agents and not agents_md:
388            return []
389
390        skill_list = "\n".join(
391            f"- {s.name}: {s.description}" for s in skills
392        ) or "(none discovered)"
393        agent_list = "\n".join(
394            f"- {a.name}: {a.description}" for a in agents
395        ) or "(none discovered)"
396
397        tasks_text = "\n".join(f"{i + 1}. {t}" for i, t in enumerate(tasks))
398
399        match_prompt = (
400            "You are a capability-matching assistant. Given a user objective, a list of tasks, "
401            "and the available skills/agents (from agents.md and discovered skill/agent files), "
402            "decide which skill or agent should be loaded to help complete each task.\n\n"
403            f"Objective: {objective}\n\n"
404            f"Tasks:\n{tasks_text}\n\n"
405            f"Available Skills:\n{skill_list}\n\n"
406            f"Available Agents:\n{agent_list}\n\n"
407            f"agents.md content:\n{agents_md or '(none found)'}\n\n"
408            "For each task, reply with exactly one line per task in this format:\n"
409            "TASK <index>: LOAD <type> <name>\n"
410            "or if no capability helps:\n"
411            "TASK <index>: NONE\n"
412            "Only match when the skill/agent name or description clearly applies to the task. "
413            "Do not invent capabilities. type is either 'skill' or 'agent'."
414        )
415        try:
416            response = await self.provider.complete(
417                system="You are a precise capability-matching assistant. Reply only in the requested format.",
418                messages=[Message.user(match_prompt)],
419                tools=[],
420            )

422                block.text for block in response.blocks if isinstance(block, TextBlock)
423            ).strip()
424        except Exception:
425            return []

427        matches: list[dict[str, Any]] = []

429        agent_by_name = {a.name: a for a in agents}
430
431        for line in result.splitlines():
432            line = line.strip()
433            if not line.startswith("TASK "):
434                continue
435            try:
436                prefix, rest = line.split(":", 1)
437                idx_str = prefix[len("TASK "):].strip()
438                task_index = int(idx_str) - 1
439                rest = rest.strip()
440                if rest == "NONE":
441                    continue

443                    parts = rest[len("LOAD "):].split(None, 1)
444                    if len(parts) != 2:
445                        continue
446                    ctype, cname = parts[0], parts[1].strip()
447                    # Skip capabilities already loaded into the main context
448                    # window (deduplication — prevents re-loading and scope creep).
449                    if f"{ctype}:{cname}" in self.loaded_capabilities:

451                    capability = None
452                    if ctype == "skill":
453                        capability = skill_by_name.get(cname)
454                    elif ctype == "agent":
455                        capability = agent_by_name.get(cname)
456                    if capability is not None:
457                        matches.append(
458                            {
459                                "type": ctype,
460                                "name": cname,
461                                "task_index": task_index,
462                                "description": getattr(capability, "description", ""),
463                            }
464                        )
465            except (ValueError, IndexError):
466                continue
467        return matches

142    async def normalize_input(self, user_input: str) -> str:
143        """Normalize user input end-to-end.
144
145        1. Refurbish the raw request via the model provider.
146        2. Split the result into objective + tasks.
147        3. Detect gaps in each task and ask the user (with suggestions) via ask_user.
148        4. Take user input once all gaps are filled.
149        5. Preserve the original input, refurbish the requirement, build a task object,
150           and return it all as the normalized user input string.
151        """
152        assert self.provider is not None
153        self.end_stream()
154
155        # Step 1: refurbish / normalize via the provider.
156        response = await self.provider.complete(
157            system=self.config.normalizer_system_prompt,
158            messages=[
159                Message.user(
160                    build_user_input_refurbish_prompt(
161                        user_input, user_input_refurbish_prompt=self.config.user_input_refurbish_prompt
162                    )
163                )
164            ],
165            tools=[],
166        )
167        normalized_text = "".join(
168            block.text for block in response.blocks if isinstance(block, TextBlock)
169        )
170        self.end_stream()
171        if normalized_text.startswith("Error:"):
172            self._ui.normalization_failed(normalized_text)
173            return normalized_text
174
175        # Non-interactive: return the refurbished text as-is.
176        if not self.interactive:
177            self._ui.input_normalized(normalized_text)
178            return normalized_text
179
180        # Step 2: split into objective + tasks.
181        objective, raw_tasks = self._split_normalized(normalized_text)
182        if not raw_tasks:
183            self._ui.input_normalized(normalized_text)
184            return normalized_text
185
186        # Step 3: build canonical task titles (strip any " — answer" enrichment
187        # from prior calls so we match against confirmed decisions).
188        tasks: list[str] = []
189        for t in raw_tasks:
190            canonical = self._canonical_task_title(t)
191            if canonical not in tasks:
192                tasks.append(canonical)
193
194        # Step 4: skip tasks whose gap was already resolved (don't re-ask,
195        # don't re-detect — prevents recursive gaps and duplicate questions).
196        pending_tasks = [t for t in tasks if t not in self.resolved_gaps]
197        if not pending_tasks:
198            self._ui.input_normalized(normalized_text)
199            return normalized_text
200
201        # Step 5: detect actual gaps only against the ORIGINAL objective and
202        # canonical task titles — never against gap descriptions, answers, or
203        # generated tasks (prevents recursive gaps).
204        gaps: list[dict[str, Any]] = []
205        for task in pending_tasks:
206            gap, suggestions = await self._detect_gap(objective, task)
207            if gap:
208                gaps.append({"task": task, "gap": gap, "suggestions": suggestions})
209
210        # Step 6: if no real gaps, continue with normalized input unchanged.
211        if not gaps:
212            self._ui.input_normalized(normalized_text)
213            return normalized_text
214
215        # Step 7: clarify each gap one-by-one with suggestions.
216        clarified_tasks: list[dict[str, str]] = []
217        for task in tasks:
218            matching_gap = None
219            for g in gaps:
220                if g["task"] == task:
221                    matching_gap = g
222                    break
223            if matching_gap is None:
224                clarified_tasks.append({"title": task, "status": "pending"})
225                continue
226
227            gap = matching_gap["gap"]
228            suggestions = matching_gap["suggestions"]
229            self._ui.console.print(f"\n[ui.subheader]Gap found for task:[/ui.subheader] {task}")
230            self._ui.console.print(f"[status.warning]Missing/unclear:[/status.warning] {gap}")
231            if suggestions:
232                self._ui.console.print(f"Suggestions: {', '.join(suggestions)}")
233
234            suggestion_text = ", ".join(suggestions) if suggestions else "please clarify"
235            question = (
236                f"Task '{task}' has a gap: {gap}. "
237                f"Did you mean {suggestion_text}? Or provide more details."
238            )
239            answer = await self.ask_user(question, suggestions or None)
240            # Preserve the confirmed decision keyed by canonical task title so
241            # this question is never asked again. Do NOT enrich the task title
242            # with the answer (prevents recursive gap detection).
243            self.confirmed_decisions[task] = (answer or "").strip()
244            self.resolved_gaps.add(task)
245            clarified_tasks.append({"title": task, "status": "pending"})
246
247        # Step 8: check existing skills/agents — internal only, never added as
248        # executable tasks (prevents scope expansion and task contamination).
249        matched_capabilities = await self._match_existing_capabilities(objective, tasks)
250        self._register_loaded_capabilities(matched_capabilities)
251
252        # Step 9: reference the normalized input with answers preserved in
253        # confirmed_decisions, but task titles stay clean.
254        final_text = self._assemble_normalized(
255            user_input, objective, clarified_tasks, matched_capabilities
256        )
257        self._ui.input_normalized(final_text)
258        return final_text
